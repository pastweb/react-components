# Walktour

Walktour guides users through a short ordered set of interface targets. Mount it
through a portal so its fixed overlay is outside the view being highlighted.

## Portal installation

Install portals once with `PortalsProvider`, or install `installPortals(...)`
on `GlobalContext`. As portal content mounts in a separate React root, install
the same portal context in the entry provider too.

## React usage

```tsx
import { createEntry, Portal, PortalsProvider, usePortal, useRef } from '@pastweb/react';
import { Button } from '@/react/components/Button';
import { Walktour, type WalktourStep } from '@/react/components/Walktour';

const anchorsIds = { walktour: 'app-walktour' };

function TourLauncher() {
  const tour = usePortal();
  const saveButton = useRef<HTMLElement | null>(null);
  const steps: WalktourStep[] = [
    { ref: saveButton, title: 'Save changes', description: 'Select Save to publish your updates.', placement: 'top' },
  ];

  return (
    <>
      <Button ref={saveButton} onClick={() => tour.open()}>Save changes</Button>
      <Portal path='walktour' use={tour}>
        <Walktour open steps={steps} onOpenChange={open => { if (!open) tour.remove(); }} />
      </Portal>
      <div id={anchorsIds.walktour} />
    </>
  );
}

function getEntry() {
  return createEntry({
    Providers: ({ children }) => <PortalsProvider anchorsIds={anchorsIds} getEntry={getEntry}>{children}</PortalsProvider>,
  });
}

export function MyComponent() {
  return <PortalsProvider anchorsIds={anchorsIds} getEntry={getEntry}><TourLauncher /></PortalsProvider>;
}
```

## Configurations

### Per-step behavior

Steps can override placement, labels, mask geometry, and navigation behavior.

```tsx
const steps: WalktourStep[] = [
  { ref: welcome, title: 'Welcome', description: 'Start here.', placement: 'bottom' },
  { ref: actions, title: 'Actions', description: 'Click the target to continue.', nextOnTargetClick: true, placement: 'right' },
  { title: 'Complete', description: 'You are ready.', placement: 'center', nextLabel: 'Finish' },
];
```

### Mask interaction

```tsx
<Walktour
  open
  steps={steps}
  closeOnMaskClick
  disableMaskInteraction={false}
  maskPadding={12}
  maskRadius={8}
/>
```
